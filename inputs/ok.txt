int somarValores(int a, int b) {
    int resultado;
    resultado = a + b;
    return resultado;
}

float calcularMedia(float soma_total, int quantidade) {
    float media;
    media = soma_total / quantidade;
    return media;
}

int verificarPar(int numero) {
    int resto;
    resto = numero % 2;
    if (resto == 0) {
        return 1;
    } else {
        return 0;
    }
}

int calcularFatorial(int n) {
    int resultado;
    int contador;
    resultado = 1;
    contador = 2;
    while (contador <= n) {
        resultado = resultado * contador;
        contador = contador + 1;
    }
    return resultado;
}
