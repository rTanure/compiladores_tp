int TOTAL;
float TAXA;

int calcularFatorial(int n) {
    if (n <= 1) {
        return 1;
    }
    return n * calcularFatorial(n - 1);
}

float calcularMedia(float soma, int quantidade) {
    return soma / quantidade;
}

void exibirTotal(int valor) {
    TOTAL = valor;
}

int verificarPar(int numero) {
    if (numero % 2 == 0) {
        return 1;
    } else {
        return 0;
    }
}

int principal() {
    int i;
    int pares;
    float media;

    TAXA = 2;
    i = 0;
    pares = 0;
    while (i < 10 && pares != 3) {
        if (verificarPar(i) == 1) {
            pares = pares + 1;
        } else {
            if (i >= 5 || !(i < 2)) {
                TOTAL = TOTAL - 1;
            }
        }
        i = i + 1;
    }

    media = calcularMedia(TAXA * 4.0, pares);
    exibirTotal(calcularFatorial(5));
    return -pares;
}
